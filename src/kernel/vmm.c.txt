#include <stdint.h>
#include <stdbool.h>
#include <mem.h>


#define PAGE_SIZE 0x1000 // 4KB
#define PAGE_TABLE_ENTRIES 0x200000 // 2 million entries
#define KERNEL_VIRTUAL_BASE 0xFFFFFFFF80000000
#define KERNEL_PHYSICAL_BASE 0x0

typedef struct {
    uint64_t present : 1;
    uint64_t writable : 1;
    uint64_t user : 1;
    uint64_t accessed : 1;
    uint64_t dirty : 1;
    uint64_t reserved : 7;
    uint64_t page_frame_number : 52;
} page_table_entry_t;

page_table_entry_t page_tables[PAGE_TABLE_ENTRIES];

void map_page(uint64_t virtual_address, uint64_t physical_address, bool user_accessible) {
    uint64_t page_table_index = (virtual_address >> 12) & 0x1FFFFF;
    page_table_entry_t* pte = &page_tables[page_table_index];
    pte->present = 1;
    pte->writable = 1;
    pte->user = user_accessible ? 1 : 0;
    pte->page_frame_number = (physical_address >> 12);
}

void* alloc_virtual_page(bool user_accessible) {
    void* physical_page = alloc_physical_page(PAGE_SIZE);
    if (physical_page) {
        uint64_t virtual_address = user_accessible ? (uint64_t)physical_page : (KERNEL_VIRTUAL_BASE + (uint64_t)physical_page - KERNEL_PHYSICAL_BASE);
        map_page(virtual_address, (uint64_t)physical_page, user_accessible);
        return (void*)virtual_address;
    }
    return NULL;
}

void free_virtual_page(void* page) {
    uint64_t virtual_address = (uint64_t)page;
    uint64_t page_table_index = (virtual_address >> 12) & 0x1FFFFF;
    page_table_entry_t* pte = &page_tables[page_table_index];
    pte->present = 0;
    free_physical_page((void*)(pte->page_frame_number << 12));
}
