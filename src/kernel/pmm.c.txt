#include <stdint.h>
#include <stddef.h>
#include <stdbool.h>
#include <mem.h>



#define PHYSICAL_MEMORY_SIZE 0x10000000 // 256MB
#define MIN_BLOCK_SIZE 0x1000 // 4KB

typedef struct memory_block {
    struct memory_block* next;
    struct memory_block* prev;
    size_t size;
    bool is_free;
    void* start_address;
} memory_block_t;

memory_block_t* memory_block_list = NULL;

void init_pmm() {
    memory_block_t* initial_block = malloc(sizeof(memory_block_t));
    initial_block->next = NULL;
    initial_block->prev = NULL;
    initial_block->size = PHYSICAL_MEMORY_SIZE;
    initial_block->is_free = true;
    initial_block->start_address = (void*)0x0;
    memory_block_list = initial_block;
}

void* alloc_physical_page(size_t size) {
    size_t adjusted_size = (size + MIN_BLOCK_SIZE - 1) / MIN_BLOCK_SIZE * MIN_BLOCK_SIZE;
    memory_block_t* current = memory_block_list;

    while (current != NULL) {
        if (current->is_free && current->size >= adjusted_size) {
            // Split the block if it's larger than the requested size
            if (current->size > adjusted_size + MIN_BLOCK_SIZE) {
                memory_block_t* new_block = malloc(sizeof(memory_block_t));
                new_block->next = current->next;
                new_block->prev = current;
                new_block->size = current->size - adjusted_size;
                new_block->is_free = true;
                new_block->start_address = (void*)((uintptr_t)current->start_address + adjusted_size);
                current->next = new_block;
                current->size = adjusted_size;
            }

            current->is_free = false;
            return current->start_address;
        }
        current = current->next;
    }

    return NULL; // No free space available
}

void free_physical_page(void* page) {
    memory_block_t* current = memory_block_list;

    while (current != NULL) {
        if (current->start_address == page) {
            current->is_free = true;

            // Merge with the next block if it's also free
            if (current->next != NULL && current->next->is_free) {
                current->size += current->next->size;
                current->next = current->next->next;
                if (current->next != NULL) {
                    current->next->prev = current;
                }
                free(current->next);
            }

            // Merge with the previous block if it's also free
            if (current->prev != NULL && current->prev->is_free) {
                current->prev->size += current->size;
                current->prev->next = current->next;
                if (current->next != NULL) {
                    current->next->prev = current->prev;
                }
                free(current);
            }

            return;
        }
        current = current->next;
    }

    // Page not found in the memory block list
    assert(false);
}
